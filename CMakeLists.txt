cmake_minimum_required(VERSION 3.20)
project(CueFrame VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CUEFRAME_VISION "Build the OpenCV camera/replay application" ON)
option(CUEFRAME_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(cueframe src/core.cpp)
target_include_directories(cueframe PUBLIC include)
target_link_libraries(cueframe PUBLIC Threads::Threads)
target_compile_options(cueframe PRIVATE -Wall -Wextra -Wpedantic)
if(CUEFRAME_SANITIZE)
  target_compile_options(cueframe PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(cueframe PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(cueframe_tests tests/core_test.cpp)
target_link_libraries(cueframe_tests PRIVATE cueframe)
add_test(NAME core_contracts COMMAND cueframe_tests)
set_tests_properties(core_contracts PROPERTIES TIMEOUT 30)
if(CUEFRAME_VISION)
  find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn)
  add_library(cueframe_vision src/detector.cpp)
  target_include_directories(cueframe_vision PUBLIC ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(cueframe_vision PUBLIC cueframe ${OpenCV_LIBS})
  target_compile_options(cueframe_vision PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(cueframe_run apps/run.cpp)
  target_link_libraries(cueframe_run PRIVATE cueframe_vision)
  add_executable(cueframe_vision_tests tests/vision_test.cpp)
  target_link_libraries(cueframe_vision_tests PRIVATE cueframe_vision)
  if(EXISTS "${CMAKE_SOURCE_DIR}/models/yolox_s.onnx")
    add_test(NAME real_model_geometry COMMAND cueframe_vision_tests
      "${CMAKE_SOURCE_DIR}/models/yolox_s.onnx" "${CMAKE_SOURCE_DIR}/examples/coffee.png")
    set_tests_properties(real_model_geometry PROPERTIES TIMEOUT 60)
  endif()
endif()
